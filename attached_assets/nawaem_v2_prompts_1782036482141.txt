برومبت 1 — إعادة تنظيم واجهة المشرف بتبويبات
الملف: NawaemAppComponents.kt → SupervisorDashboardScreen
══════════════════════════════════════════════════

في SupervisorDashboardScreen في NawaemAppComponents.kt
أضف متغير تبويب في أعلى الدوال:
var selectedTab by remember { mutableStateOf(0) }

ثم في body الـ Scaffold بعد TopBar مباشرة أضف:
TabRow(
    selectedTabIndex = selectedTab,
    containerColor = MaterialTheme.colorScheme.surface,
    contentColor = MaterialTheme.colorScheme.primary
) {
    Tab(
        selected = selectedTab == 0,
        onClick = { selectedTab = 0 },
        text = { Text("📋 تحضير الموظفين", fontWeight = FontWeight.Bold) }
    )
    Tab(
        selected = selectedTab == 1,
        onClick = { selectedTab = 1 },
        text = { Text("⚙️ الإعدادات والتذكير", fontWeight = FontWeight.Bold) }
    )
}

ثم اعمل الآتي:
- كل الكود المتعلق بـ "تحضير العمالة والموظفين الميدانيين اليومي"
  والنصوص التوضيحية والتذكيرات (تذكير الصباح / تذكير المساء)
  وكتلة المحاكاة (على الوقت / 15 دقيقة / 45 دقيقة / +2 ساعة)
  وحقل الملاحظات → ضعها في if (selectedTab == 1) { ... }

- كل الكود المتعلق ببطاقات الموظفين وأزرار الحضور
  وزر اعتماد التحضير → ضعها في if (selectedTab == 0) { ... }

- في التبويب 0 أبقِ فقط في الأعلى:
  → اختيار الوردية (صباحي/مسائي)
  → التاريخ المحدد
  → رسالة القفل إذا كان محقوناً
  → بطاقات الموظفين مباشرة بدون أي نصوص توضيحية

لا تحذف أي منطق، فقط أعد التوزيع.
ادفع التغييرات فقط.

برومبت 2 — التحضير الاسترجاعي (اختيار تاريخ ماضٍ)
الملف: NawaemAppComponents.kt → SupervisorDashboardScreen
══════════════════════════════════════════════════

في SupervisorDashboardScreen في NawaemAppComponents.kt
حوّل التاريخ من ثابت إلى قابل للتغيير:

استبدل هذا:
val formattedDate = remember { sdfDate.format(java.util.Date()) }

بهذا:
var selectedDate by remember { mutableStateOf(sdfDate.format(java.util.Date())) }
val todayDate = remember { sdfDate.format(java.util.Date()) }
var showDatePicker by remember { mutableStateOf(false) }

ثم أضف في التبويب 0 (تبويب التحضير) زراً لاختيار التاريخ:
Row(
    modifier = Modifier.fillMaxWidth().padding(horizontal = 8.dp),
    horizontalArrangement = Arrangement.SpaceBetween,
    verticalAlignment = Alignment.CenterVertically
) {
    TextButton(onClick = { showDatePicker = true }) {
        Icon(Icons.Default.DateRange, contentDescription = null)
        Spacer(Modifier.width(4.dp))
        Text("تغيير التاريخ", fontSize = 12.sp)
    }
    Text(
        text = if (selectedDate == todayDate) "📅 اليوم - $selectedDate"
               else "📅 $selectedDate (استرجاعي)",
        fontWeight = FontWeight.Bold,
        color = if (selectedDate == todayDate) MaterialTheme.colorScheme.primary
                else Color(0xFFE65100),
        fontSize = 13.sp
    )
}

ثم أضف DatePickerDialog بسيط باستخدام AlertDialog:
if (showDatePicker) {
    val calendar = remember { java.util.Calendar.getInstance() }
    var pickedYear by remember { mutableStateOf(calendar.get(java.util.Calendar.YEAR)) }
    var pickedMonth by remember { mutableStateOf(calendar.get(java.util.Calendar.MONTH)) }
    var pickedDay by remember { mutableStateOf(calendar.get(java.util.Calendar.DAY_OF_MONTH)) }

    AlertDialog(
        onDismissRequest = { showDatePicker = false },
        title = { Text("اختر تاريخ التحضير") },
        text = {
            Column {
                Text("اليوم من 1 إلى ${calendar.getActualMaximum(java.util.Calendar.DAY_OF_MONTH)}")
                Slider(
                    value = pickedDay.toFloat(),
                    onValueChange = { pickedDay = it.toInt() },
                    valueRange = 1f..calendar.getActualMaximum(java.util.Calendar.DAY_OF_MONTH).toFloat(),
                    steps = calendar.getActualMaximum(java.util.Calendar.DAY_OF_MONTH) - 2
                )
                Text("اليوم المحدد: $pickedDay / ${pickedMonth + 1} / $pickedYear",
                    fontWeight = FontWeight.Bold,
                    color = MaterialTheme.colorScheme.primary)
            }
        },
        confirmButton = {
            Button(onClick = {
                val cal = java.util.Calendar.getInstance()
                cal.set(pickedYear, pickedMonth, pickedDay)
                val today = java.util.Calendar.getInstance()
                if (!cal.after(today)) {
                    selectedDate = sdfDate.format(cal.time)
                }
                showDatePicker = false
            }) { Text("تأكيد") }
        },
        dismissButton = {
            TextButton(onClick = { showDatePicker = false }) { Text("إلغاء") }
        }
    )
}

ثم استبدل كل مكان يستخدم formattedDate بـ selectedDate
واستبدل كل مكان يستخدم formattedDay بـ:
remember(selectedDate) {
    sdfDay.format(sdfDate.parse(selectedDate) ?: java.util.Date())
}

ادفع التغييرات فقط ولا تمس أي كود آخر.
